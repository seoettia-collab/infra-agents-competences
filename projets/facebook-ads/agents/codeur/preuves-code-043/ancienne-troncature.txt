        // Fonction de troncature intelligente (coupe au dernier mot complet)
        function smartTruncate(text, maxLen) {
            if (!text || text.length <= maxLen) return text;
            // D'abord essayer sans emojis
            const noEmoji = text.replace(/[\u{1F600}-\u{1F9FF}]|[\u{2600}-\u{27BF}]|[\u{1F300}-\u{1F5FF}]|[\u{1F680}-\u{1F6FF}]/gu, '').trim();
            if (noEmoji.length <= maxLen) return noEmoji;
            // Couper au dernier espace avant la limite
            const cut = noEmoji.substring(0, maxLen);
            const lastSpace = cut.lastIndexOf(' ');
            return lastSpace > maxLen * 0.6 ? cut.substring(0, lastSpace) : cut;
        }
